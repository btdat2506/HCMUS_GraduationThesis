\section{Test Program Documentation and Configurations}

This section documents the specific configuration of each test program used in the evaluation, ensuring full reproducibility of the results.

\subsection{Matrix Multiplication Tests (bareMetalC/)}

\subsubsection{Basic Matrix Multiplication Tests}
\begin{itemize}
\item \texttt{tiled\_matmul\_os}: Output-stationary matrix multiplication
  \begin{itemize}
  \item Uses Gemmini's output-stationary dataflow
  \item Matrix dimensions: 512$\times$512$\times$512 (non-baremetal)
  \end{itemize}

\item \texttt{tiled\_matmul\_ws}: Weight-stationary matrix multiplication
  \begin{itemize}
  \item Uses Gemmini's weight-stationary dataflow (most efficient)
  \item Matrix dimensions: 512$\times$512$\times$512 (I$\times$J$\times$K)
  \end{itemize}
\end{itemize}

\subsubsection{Matrix Multiplication Variants}
\begin{itemize}
\item \texttt{tiled\_matmul\_ws\_At}: Weight-stationary with A-transpose
  \begin{itemize}
  \item Same as \texttt{tiled\_matmul\_ws} but with matrix A transposed
  \item Matrix dimensions: 500$\times$300$\times$412 (I$\times$J$\times$K)
  \end{itemize}

\item \texttt{tiled\_matmul\_ws\_Bt}: Weight-stationary with B-transpose
  \begin{itemize}
  \item Same as \texttt{tiled\_matmul\_ws} but with matrix B transposed
  \end{itemize}

\item \texttt{tiled\_matmul\_ws\_full\_C}: Weight-stationary with full accumulator width
  \begin{itemize}
  \item Uses full-width accumulator for output matrix C
  \end{itemize}

\item \texttt{tiled\_matmul\_ws\_low\_D}: Weight-stationary with bias matrix
  \begin{itemize}
  \item Includes bias matrix D in computation (C = A$\times$B + D)
  \end{itemize}
\end{itemize}

\subsubsection{Performance Analysis Tests}
\begin{itemize}
\item \texttt{tiled\_matmul\_ws\_perf}: Weight-stationary performance analysis
  \begin{itemize}
  \item Matrix dimensions: 128$\times$256$\times$512 (I$\times$J$\times$K)
  \item \textbf{Verified metrics}: 
    \begin{itemize}
    \item Total MACs: 16,777,216
    \item Time Execution (ms): 2.426 | \textbf{Utilization: 2\%}
    \item Memory traffic: 2.16MB RDMA reads, 32KB WDMA writes
    \end{itemize}
  \end{itemize}
\end{itemize}

\subsection{Convolution Tests (bareMetalC/)}

\subsubsection{Standard Convolution Tests}
\begin{itemize}
\item \texttt{conv}: Basic 2D convolution operation
  \begin{itemize}
  \item Input: 224$\times$224$\times$3 (batch=4), Output: 112$\times$112$\times$32
  \item Kernel: 3$\times$3, Stride: 2, Padding: 1
  \end{itemize}

\item \texttt{conv\_first\_layer}: First layer convolution (typical CNN entry)
  \begin{itemize}
  \item Same configuration as \texttt{conv} but optimized for first layer
  \item Input: 224$\times$224$\times$3 (batch=4), Output: 112$\times$112$\times$32
  \end{itemize}

\item \texttt{conv\_rect}: Rectangular convolution
  \begin{itemize}
  \item Tests non-square convolution operations
  \item Input: 224$\times$224$\times$3 (batch=4), Output: 112$\times$112$\times$32
  \end{itemize}

\item \texttt{conv\_stride}: Strided convolution
  \begin{itemize}
  \item Tests convolution with stride=2
  \item Input: 224$\times$224$\times$3 (batch=4), Output: 112$\times$112$\times$32
  \end{itemize}
\end{itemize}

\subsubsection{Depthwise Convolution Tests}
\begin{itemize}
\item \texttt{conv\_dw}: Depthwise convolution
  \begin{itemize}
  \item Input: 112$\times$112$\times$17 (batch=3), Output: 56$\times$56$\times$17
  \item Lower computational intensity than standard convolution
  \end{itemize}
\end{itemize}

\subsubsection{Transformed Convolution Tests}
\begin{itemize}
\item \texttt{conv\_trans\_output\_1203}: Output transformation (1,2,0,3)
  \begin{itemize}
  \item Tests convolution with output tensor transformation
  \item Input: 224$\times$224$\times$17 (batch=4), Output: 112$\times$112$\times$32
  \end{itemize}

\item \texttt{conv\_trans\_weight\_0132}: Weight transformation (0,1,3,2)
  \begin{itemize}
  \item Tests convolution with weight tensor transformation
  \item Input: 224$\times$224$\times$17 (batch=4), Output: 112$\times$112$\times$32
  \end{itemize}

\item \texttt{conv\_trans\_weight\_1203}: Weight transformation (1,2,0,3)
  \begin{itemize}
  \item Alternative weight tensor transformation
  \item Input: 224$\times$224$\times$17 (batch=4), Output: 112$\times$112$\times$32
  \end{itemize}
\end{itemize}

\subsubsection{Convolution with Pooling Tests}
\begin{itemize}
\item \texttt{conv\_with\_pool}: Convolution followed by pooling
  \begin{itemize}
  \item Input: 224$\times$224$\times$3 (batch=4), Conv output: 112$\times$112$\times$32, Pool output: 56$\times$56$\times$32
  \end{itemize}

\item \texttt{conv\_rect\_pool}: Rectangular convolution with pooling
  \begin{itemize}
  \item Same configuration as \texttt{conv\_with\_pool}
  \end{itemize}
\end{itemize}

\subsubsection{Special Convolution Tests}
\begin{itemize}
\item \texttt{conv\_with\_rot180}: Convolution with 180° rotation
  \begin{itemize}
  \item Tests convolution with rotated input
  \item Input: 224$\times$224$\times$3 (batch=4), Output: 224$\times$224$\times$17 (stride=1, no size reduction)
  \end{itemize}
\end{itemize}

\subsection{Neural Network Models (imagenet/)}
\begin{itemize}
\item \texttt{mobilenet}: Complete MobileNet v1 inference
  \begin{itemize}
  \item Architecture: Standard convolution + 13 depthwise separable convolution blocks
  \item Input: 224$\times$224$\times$3 ImageNet images
  \end{itemize}
\end{itemize}

\subsection{Multi-Layer Perceptron Models (mlps/)}
\begin{itemize}
\item \texttt{mlp1}: 6-layer MLP with RELU activation
  \begin{itemize}
  \item Architecture: 832$\rightarrow$2560$\rightarrow$2048$\rightarrow$1536$\rightarrow$1024$\rightarrow$512$\rightarrow$64
  \item Original dimensions: 784$\rightarrow$2500$\rightarrow$2000$\rightarrow$1500$\rightarrow$1000$\rightarrow$500$\rightarrow$10 (zero-padded)
  \item Batch size: 64
  \item \textbf{Total time execution (ms)}: 170.637
  \item \textbf{Layer breakdown}: 22.4M, 77.3M, 45.7M, 18.7M, 6.2M, 0.5M (ms)
  \end{itemize}

\item \texttt{mlp1\_32}: 32-bit version of MLP1
  \begin{itemize}
  \item Same architecture as \texttt{mlp1} but with 32-bit precision
  \item \textbf{Total Time Execution (ms)}: 174.870
  \item \textbf{Layer breakdown}: 22.4M, 77.1M, 46.1M, 22.5M, 6.2M, 0.5M (ms)
  \end{itemize}
  
\item \texttt{mlp4}: 2-layer MLP with RELU activation
  \begin{itemize}
  \item Architecture: 3072$\rightarrow$4608$\rightarrow$3072
  \item Original dimensions: 3036$\rightarrow$4554$\rightarrow$3036 (zero-padded)
  \item Batch size: 64
  \item \textbf{Total Time Execution (ms)}: 432.838
  \item \textbf{Layer breakdown}: 215.6M, 217.2M (ms)
  \end{itemize}
\end{itemize}
